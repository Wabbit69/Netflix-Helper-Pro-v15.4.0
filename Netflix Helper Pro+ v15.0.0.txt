    /* ─────────────────────────────────────────────────────────────
       NETWORK TRACKER

    class NetTracker {
        constructor() {
            this.events = [];
            this.history = [];
            this.sawEntry = false;
            this.sawSize = false;
            this.t0 = performance.now();
            this._obs = null;
            this._start();
        }

        _start() {
            if (typeof PerformanceObserver === 'undefined') return;
            try {
                this._obs = new PerformanceObserver(list => {
                    const now = performance.now();
                    for (const e of list.getEntries()) {
                        if (e.initiatorType !== 'fetch' && e.initiatorType !== 'xmlhttprequest') continue;
                        if (!/nflxvideo\.net|nflxso\.net/i.test(e.name)) continue;
                        this.sawEntry = true;
                        const bytes = e.transferSize || e.encodedBodySize || 0;
                        if (bytes > 0) { this.sawSize = true; this.events.push({ t: now, bytes }); }
                    }
                });
                this._obs.observe({ type: 'resource', buffered: false });
            } catch (e) { warn('PerformanceObserver unavailable:', e); }
        }

        get blocked() { return this.sawEntry && !this.sawSize; }

        rate(windowMs = CFG.netWindow) {
            const now = performance.now();
            const cut = now - windowMs;
            while (this.events.length && this.events[0].t < cut) this.events.shift();
            if (!this.events.length) return 0;
            const span = Math.min(windowMs, Math.max(1000, now - this.t0)) / 1000;
            const bytes = this.events.reduce((a, e) => a + e.bytes, 0);
            return span > 0 ? (bytes * 8) / span / 1e6 : 0;
        }

        sample() {
            const m = this.rate();
            if (m > 0) {
                this.history.push(m);
                if (this.history.length > 300) this.history.shift();
            }
            return m;
        }

        avg() {
            return this.history.length ? this.history.reduce((a, b) => a + b, 0) / this.history.length : 0;
        }

        reset() { this.events = []; this.history = []; this.t0 = performance.now(); }

        destroy() {
            if (this._obs) { try { this._obs.disconnect(); } catch { /* ignore */ } this._obs = null; }
        }
    }

    class FpsMeter {
        constructor() {
            this.fps = 0;
            this._v = null;
            this._id = 0;
            this._active = false;
            this._lastMedia = 0;
            this._lastFrames = 0;
        }

        sync(video, shouldRun) {
            if (!shouldRun || !video || typeof video.requestVideoFrameCallback !== 'function') {
                this.stop();
                return;
            }
            if (this._active && this._v === video) return;
            this.stop();
            this._v = video;
            this._active = true;
            this._lastMedia = 0;
            this._lastFrames = 0;

            const step = (now, meta) => {
                if (!this._active || this._v !== video || now < 0) return;
                if (this._lastFrames > 0) {
                    const df = meta.presentedFrames - this._lastFrames;
                    const dm = meta.mediaTime - this._lastMedia;
                    if (dm >= 0.5 && df > 0 && dm < 5) {
                        const est = df / dm;
                        if (est > 5 && est < 140) this.fps = est;
                        this._lastMedia = meta.mediaTime;
                        this._lastFrames = meta.presentedFrames;
                    } else if (dm < 0 || dm >= 5) {
                        this._lastMedia = meta.mediaTime;
                        this._lastFrames = meta.presentedFrames;
                    }
                } else {
                    this._lastMedia = meta.mediaTime;
                    this._lastFrames = meta.presentedFrames;
                }
                this._id = video.requestVideoFrameCallback(step);
            };
            this._id = video.requestVideoFrameCallback(step);
        }

        stop() {
            if (this._v && this._id && typeof this._v.cancelVideoFrameCallback === 'function') {
                try { this._v.cancelVideoFrameCallback(this._id); } catch { /* ignore */ }
            }
            this._active = false;
            this._v = null;
            this._id = 0;
            this.fps = 0;
        }
    }

    /* ─────────────────────────────────────────────────────────────
       REACTIVE AMBILIGHT & BIAS HALO ENGINE
       - Paints a blurred ambient projection glow behind the video.
       - Automatically detects if hardware DRM blanks canvas readback and
         falls back to zero-CPU warm cinema D65 bias lighting.
    ───────────────────────────────────────────────────────────── */
    class AmbiEngine {
        constructor() {
            this.canvas = null;
            this.ctx = null;
            this._timer = 0;
            this._drmBlocked = false;
            this._darkHits = 0;
            this._v = null;
        }

        _ensureCanvas(video) {
            const parent = video.parentElement;
            if (!parent) return false;
            if (!this.canvas) {
                const c = document.createElement('canvas');
                c.className = 'nfp-ambi';
                c.width = 32;
                c.height = 18;
                this.canvas = c;
                this.ctx = c.getContext('2d', { willReadFrequently: true });
            }
            if (this.canvas.parentElement !== parent) {
                parent.insertBefore(this.canvas, video);
            }
            return true;
        }

        _paintBiasGlow() {
            if (!this.ctx) return;
            const g = this.ctx.createRadialGradient(16, 9, 2, 16, 9, 18);
            g.addColorStop(0, 'rgba(255, 198, 135, 0.55)');
            g.addColorStop(0.6, 'rgba(180, 120, 75, 0.32)');
            g.addColorStop(1, 'rgba(20, 14, 10, 0)');
            this.ctx.clearRect(0, 0, 32, 18);
            this.ctx.fillStyle = g;
            this.ctx.fillRect(0, 0, 32, 18);
        }

        sync(enabled, video) {
            if (!enabled || !video || !isWatchPage()) {
                this.stop();
                return;
            }
            if (this._v !== video) {
                this._v = video;
                this._drmBlocked = false;
                this._darkHits = 0;
            }
            if (!this._ensureCanvas(video)) return;
            if (this._timer) return;

            this._paintBiasGlow();
            this._timer = setInterval(() => this._step(), CFG.ambiTick);
        }

        _step() {
            const v = this._v;
            if (!v || !v.isConnected || !this.ctx) { this.stop(); return; }
            if (this._drmBlocked || v.paused || v.readyState < 2 || !v.videoWidth) return;

            try {
                this.ctx.drawImage(v, 0, 0, 32, 18);
                if (this._darkHits < 8) {
                    const d = this.ctx.getImageData(4, 4, 24, 10).data;
                    let lit = false;
                    for (let i = 0; i < d.length; i += 16) {
                        if (d[i] > 10 || d[i + 1] > 10 || d[i + 2] > 10) { lit = true; break; }
                    }
                    if (!lit && v.currentTime > 2) {
                        this._darkHits++;
                        if (this._darkHits >= 6) {
                            this._drmBlocked = true;
                            this._paintBiasGlow();
                        }
                    } else if (lit) {
                        this._darkHits = 10; // Confirmed readable: never call getImageData again
                    }
                }
            } catch {
                this._drmBlocked = true;
                this._paintBiasGlow();
            }
        }

        stop() {
            if (this._timer) { clearInterval(this._timer); this._timer = 0; }
            if (this.canvas) { this.canvas.remove(); }
            this._v = null;
        }
    }

    class SubTracker {
        constructor(ui) {
            this.ui = ui;
            this.lines = [];
            this.lastText = '';
            this._mo = null;
            this._deb = 0;
        }

        start() {
            this.stop();
            if (!document.body) return;
            this._mo = new MutationObserver(() => {
                if (!isWatchPage() || this._deb) return;
                this._deb = setTimeout(() => { this._deb = 0; this._scan(); }, 90);
            });
            this._mo.observe(document.body, { childList: true, subtree: true, characterData: true });
        }

        stop() {
            if (this._mo) { this._mo.disconnect(); this._mo = null; }
            if (this._deb) { clearTimeout(this._deb); this._deb = 0; }
        }

        reset() {
            this.lines = [];
            this.lastText = '';
            this.ui._renderSubLog();
        }

        currentText() {
            const nodes = document.querySelectorAll('.watch-video .player-timedtext-text-container');
            if (!nodes.length) return '';
            const parts = [];
            nodes.forEach(n => {
                const t = (n.innerText || n.textContent || '').replace(/\s+/g, ' ').trim();
                if (t) parts.push(t);
            });
            return parts.join(' ');
        }

        _scan() {
            const txt = this.currentText();
            if (!txt || txt === this.lastText) {
                if (!txt) this.lastText = '';
                return;
            }
            this.lastText = txt;
            const cur = NFPlayer.currentTime() || (DOM.video()?.currentTime ?? 0);
            const t = Math.max(0, cur - 0.15);
            const prev = this.lines[this.lines.length - 1];
            if (!prev || prev.text !== txt) {
                this.lines.push({ t, text: txt });
                if (this.lines.length > 35) this.lines.shift();
                this.ui._renderSubLog();
            }
            if (S.get('subCopy') && navigator.clipboard) {
                navigator.clipboard.writeText(txt).catch(() => {});
            }
        }

        replay() {
            if (!this.lines.length) return null;
            const now = NFPlayer.currentTime() || 0;
            let idx = this.lines.length - 1;
            if (idx > 0 && now - this.lines[idx].t < 1.3) idx--;
            const item = this.lines[idx];
            NFPlayer.seekTo(item.t);
            return item;
        }
    }

    class BrowseEnhancer {
        constructor() {

        async _lookup(title) {

            const p = (async () => {
                const q = encodeURIComponent(title);

       WATCH-TIME TRACKER

    class WatchTracker {
        constructor() {
            this.log = S.get('watchLog');
            this.last = 0;
            this.session = 0;
            this.dirty = false;
            this.lastFlush = Date.now();
            this._prune();
        }

        _prune() {
            const keys = Object.keys(this.log).sort();
            while (keys.length > 120) delete this.log[keys.shift()];
        }

        tick(playing) {
            const now = Date.now();
            const dt = this.last ? Math.min((now - this.last) / 1000, 2.5) : 0;
            this.last = now;
            if (!playing || dt <= 0) return;
            const k = dayKey();
            this.log[k] = (this.log[k] || 0) + dt;
            this.session += dt;
            this.dirty = true;
            if (now - this.lastFlush > CFG.watchFlush) this.flush();
        }

        flush() {
            if (!this.dirty) return;
            S.set('watchLog', this.log);
            this.dirty = false;
            this.lastFlush = Date.now();
        }

        today() { return this.log[dayKey()] || 0; }

        days(n) {
            let sum = 0;
            const d = new Date();
            for (let i = 0; i < n; i++) {
                sum += this.log[dayKey(d)] || 0;
                d.setDate(d.getDate() - 1);
            }
            return sum;
        }

        total() { return Object.values(this.log).reduce((a, b) => a + b, 0); }

        reset() { this.log = {}; this.session = 0; this.dirty = false; S.set('watchLog', this.log); }
    }

    /* ─────────────────────────────────────────────────────────────
       AUDIO ENGINE (3D Theater Convolver + Haas Soundstage + EQ + Boost)
       - Never attaches Web Audio while boost === 100% AND mode === 'off',
         preserving untouched multichannel passthrough until needed.
    ───────────────────────────────────────────────────────────── */
    const AUDIO_PRESETS = {
        off: { hp: 12, midGain: 0, lowGain: 0, thr: -3, knee: 6, ratio: 12, makeup: 1.00, rev: 0, haas: 0 },
        theater: { hp: 28, midGain: 2.2, lowGain: 4.2, thr: -10, knee: 10, ratio: 6, makeup: 1.12, rev: 0.22, haas: 0.16 },
        voice: { hp: 95, midGain: 5.0, lowGain: -2.5, thr: -18, knee: 12, ratio: 4, makeup: 1.25, rev: 0, haas: 0 },
        night: { hp: 80, midGain: 3.0, lowGain: -4.0, thr: -28, knee: 15, ratio: 8, makeup: 1.65, rev: 0, haas: 0 },
        cinema: { hp: 20, midGain: 1.5, lowGain: 4.5, thr: -8, knee: 8, ratio: 8, makeup: 1.05, rev: 0.08, haas: 0.08 }
    };

    class AudioEngine {
        constructor() {
            this.ctx = null;
            this.impulse = null;
            this.nodes = new WeakMap();
            this.pct = 100;
            this.mode = 'off';
        }

        _ctx() {
            if (this.ctx) return this.ctx;
            const AC = window.AudioContext || window.webkitAudioContext;
            if (!AC) return null;
            try { this.ctx = new AC(); } catch { return null; }
            return this.ctx;
        }

        // Synthesize a velvet-damped auditorium impulse response (20ms pre-delay, 0.42s RT60)
        _hallImpulse(ctx) {
            if (this.impulse) return this.impulse;
            const rate = ctx.sampleRate;
            const len = Math.floor(rate * 0.42);
            const pre = Math.floor(rate * 0.02);
            const buf = ctx.createBuffer(2, len, rate);
            for (let ch = 0; ch < 2; ch++) {
                const d = buf.getChannelData(ch);
                let prev = 0;
                for (let i = pre; i < len; i++) {
                    const p = (i - pre) / (len - pre);
                    const decay = Math.pow(1 - p, 2.8);
                    const white = Math.random() * 2 - 1;
                    prev = prev * 0.64 + white * 0.36; // 1-pole HF absorption like cinema seats
                    d[i] = prev * decay * 0.48;
                }
            }
            this.impulse = buf;
            return buf;
        }

        _connect(video) {
            try {
                const ctx = this.ctx;
                const src = ctx.createMediaElementSource(video);
                const hp = ctx.createBiquadFilter();
                const mid = ctx.createBiquadFilter();
                const low = ctx.createBiquadFilter();
                const dryGain = ctx.createGain();
                const comp = ctx.createDynamicsCompressor();

                hp.type = 'highpass'; hp.Q.value = 0.7;
                mid.type = 'peaking'; mid.frequency.value = 2500; mid.Q.value = 0.95;
                low.type = 'lowshelf'; low.frequency.value = 85;
                comp.attack.value = 0.004;
                comp.release.value = 0.25;

                // Spatial theater room & Haas early-reflection network
                const wetHp = ctx.createBiquadFilter();
                wetHp.type = 'highpass'; wetHp.frequency.value = 160;
                const wetLp = ctx.createBiquadFilter();
                wetLp.type = 'lowpass'; wetLp.frequency.value = 3400;

                const conv = ctx.createConvolver();
                conv.buffer = this._hallImpulse(ctx);
                const revGain = ctx.createGain();
                revGain.gain.value = 0;

                const split = ctx.createChannelSplitter(2);
                const merge = ctx.createChannelMerger(2);
                const dL = ctx.createDelay(0.05); dL.delayTime.value = 0.013;
                const dR = ctx.createDelay(0.05); dR.delayTime.value = 0.019;
                const haasGain = ctx.createGain();
                haasGain.gain.value = 0;

                // Dry path (phase-coherent phantom center)
                src.connect(hp);
                hp.connect(mid);
                mid.connect(low);
                low.connect(dryGain);
                dryGain.connect(comp);

                // Wet spatial path (160Hz-3.4kHz warm room reflections)
                low.connect(wetHp);
                wetHp.connect(wetLp);
                wetLp.connect(conv);
                conv.connect(revGain);
                revGain.connect(comp);

                wetLp.connect(split);
                split.connect(dL, 0);
                split.connect(dR, 1);
                dL.connect(merge, 0, 1);
                dR.connect(merge, 0, 0);
                merge.connect(haasGain);
                haasGain.connect(comp);

                comp.connect(ctx.destination);

                const n = { hp, mid, low, dryGain, revGain, haasGain, comp };
                this.nodes.set(video, n);
                return n;
            } catch (e) { warn('AudioEngine could not attach:', e); return null; }
        }

        async apply(pct, mode, video) {
            this.pct = pct;
            this.mode = AUDIO_PRESETS[mode] ? mode : 'off';
            if (!video) return true;

            let n = this.nodes.get(video);
            if (!n) {
                if (pct <= 100 && this.mode === 'off') return true;
                const ctx = this._ctx();
                if (!ctx) return false;
                if (ctx.state !== 'running') { try { await ctx.resume(); } catch { /* needs gesture */ } }
                if (ctx.state !== 'running') return 'pending';
                n = this._connect(video);
                if (!n) return false;
            } else if (this.ctx && this.ctx.state === 'suspended') {
                this.ctx.resume().catch(() => {});
            }

            const p = AUDIO_PRESETS[this.mode];
            const t = this.ctx.currentTime;
            const scale = this.pct / 100;
            n.hp.frequency.setTargetAtTime(p.hp, t, 0.03);
            n.mid.gain.setTargetAtTime(p.midGain, t, 0.03);
            n.low.gain.setTargetAtTime(p.lowGain, t, 0.03);
            n.comp.threshold.setTargetAtTime(p.thr, t, 0.03);
            n.comp.knee.setTargetAtTime(p.knee, t, 0.03);
            n.comp.ratio.setTargetAtTime(p.ratio, t, 0.03);
            n.dryGain.gain.setTargetAtTime(scale * p.makeup, t, 0.03);
            n.revGain.gain.setTargetAtTime(scale * p.rev, t, 0.04);
            n.haasGain.gain.setTargetAtTime(scale * p.haas, t, 0.04);
            return true;
        }
    }

    class SleepTimer {
        constructor(ui) {
            this.ui = ui;
            this.mode = null;
            this.choice = '0';
            this.endAt = 0;
        }

        set(val) {
            val = String(val);
            if (val === 'end') {
                this.mode = 'end';
                this.choice = 'end';
                this.endAt = 0;
                return;
            }
            const m = parseInt(val, 10);
            if (m > 0) {
                this.mode = 'min';
                this.choice = String(m);
                this.endAt = Date.now() + m * 60000;
            } else {
                this.mode = null;
                this.choice = '0';
                this.endAt = 0;
            }
        }

        _timeLeft(v) {
            const d = NFPlayer.duration(), c = NFPlayer.currentTime();
            if (d > 0) return d - c;
            return isFinite(v.duration) ? v.duration - v.currentTime : Infinity;
        }

        tick(v) {
            if (!this.mode || !v) return;
            if (this.mode === 'min' && Date.now() >= this.endAt) {
                this._fire('⏰ Sleep timer — paused');
            } else if (this.mode === 'end' && !v.paused) {
                if (this._timeLeft(v) <= 1.2 + 1.5 * v.playbackRate) this._fire('😴 End of episode — paused');
            }
        }

        label() {
            if (this.mode === 'min') return `Pausing in ${fmtClock((this.endAt - Date.now()) / 1000)}`;
            if (this.mode === 'end') return 'Pausing at end of episode';
            return 'Off';
        }

        _fire(msg) {
            this.mode = null;
            this.choice = '0';
            this.endAt = 0;
            this.ui.pauseVideo();
            this.ui.toast(msg, 'sleep');
            this.ui._syncSleepUI();
        }
    }

    class AutoSkipEngine {
        constructor(ui) {
            this.ui = ui;
            this.cool = Object.create(null);
            this.until = 0;
            this._poll = null;
            this._mo = null;
            this._deb = 0;
            this._loopCool = 0;
        }

        suspend(ms) { this.until = Math.max(this.until, Date.now() + ms); }
        newEpisode() { this.cool = Object.create(null); }

        start() {
            this.stop();
            this._poll = setInterval(() => this._check(), CFG.skipPoll);
            if (document.body) {
                this._mo = new MutationObserver(() => {
                    if (this._deb) return;
                    this._deb = setTimeout(() => { this._deb = 0; this._check(); }, 120);
                });
                this._mo.observe(document.body, { childList: true, subtree: true });
            }
        }

        stop() {
            if (this._poll) { clearInterval(this._poll); this._poll = null; }
            if (this._mo) { this._mo.disconnect(); this._mo = null; }
            if (this._deb) { clearTimeout(this._deb); this._deb = 0; }
        }

        _checkLoop() {
            const ab = this.ui.ab;
            if (ab.a === null || ab.b === null || ab.b <= ab.a + 1) return;
            const now = Date.now();
            if (now - this._loopCool < 1000) return;
            const cur = NFPlayer.currentTime();
            if (cur >= ab.b) {
                this._loopCool = now;
                NFPlayer.seekTo(ab.a);
            }
        }

        _check() {
            if (!isWatchPage()) return;
            this._checkLoop();
            if (Date.now() < this.until) return;

            const intro = S.get('autoIntro');
            const next = S.get('autoNext') && this.ui.sleep.mode !== 'end';
            const credits = !next && (S.get('autoCredits') || S.get('cinemaMode'));
            const cont = S.get('autoContinue');
            if (!intro && !next && !credits && !cont) return;

            if (intro && !this._try('intro', SEL.intro, 3000, true)) {
                this._try('recap', SEL.recap, 3000, true);
            }
            if (next) this._try('next', SEL.next, 10000, false);
            if (credits) this._try('credits', SEL.credits, 8000, false);
            if (cont) this._try('cont', SEL.cont, 10000, false);
        }

        _try(kind, selectors, cooldown, measure) {
            const now = Date.now();
            if (now - (this.cool[kind] || 0) < cooldown) return false;
            const btn = DOM.firstVisible(selectors);
            if (!btn) return false;
            this.cool[kind] = now;
            const t0 = measure ? NFPlayer.currentTime() : 0;
            log('AutoSkip:', kind);
            setTimeout(() => {
                try { btn.click(); } catch { /* ignore */ }
                if (measure) this.ui.recordSkip(t0);
            }, 150);
            return true;
        }
    }

        _drag() {
            const panel = this.panel, hd = this.$('nfpHd');
            let drag = null;

            this._on(hd, 'pointerdown', e => {
                if (e.button !== 0 || e.target.closest('.nfp-btns')) return;
                const r = panel.getBoundingClientRect();
                drag = { id: e.pointerId, sx: e.clientX, sy: e.clientY, l: r.left, t: r.top };
                try { hd.setPointerCapture(e.pointerId); } catch { /* ignore */ }
                hd.classList.add('drag');
                e.preventDefault();
            });
            this._on(hd, 'pointermove', e => {
                if (!drag || e.pointerId !== drag.id) return;
                this._place(drag.l + e.clientX - drag.sx, drag.t + e.clientY - drag.sy);
            });
            const end = e => {
                if (!drag || (e && e.pointerId !== undefined && e.pointerId !== drag.id)) return;
                try { hd.releasePointerCapture(drag.id); } catch { /* ignore */ }
                drag = null;
                hd.classList.remove('drag');
                const r = panel.getBoundingClientRect();
                S.set('pos', { l: Math.round(r.left), t: Math.round(r.top) });
            };
            ['pointerup', 'pointercancel', 'lostpointercapture'].forEach(t => this._on(hd, t, end));
            this._on(window, 'blur', () => end());
            this._on(window, 'resize', () => this._clampToViewport());

            this._on(hd, 'dblclick', e => {
                if (e.target.closest('.nfp-btns')) return;
                panel.style.left = ''; panel.style.top = ''; panel.style.right = '';
                S.set('pos', null);
            });

            this._on(panel, 'mouseenter', () => { clearTimeout(this._dimT); panel.classList.remove('dim'); });
            this._on(panel, 'mouseleave', () => {
                clearTimeout(this._dimT);
                this._dimT = setTimeout(() => { if (!drag) panel.classList.add('dim'); }, CFG.dimDelay);
            });

            // Wake invisible Cinema Mode panel when cursor reaches top-right corner
            this._on(document, 'mousemove', e => {
                if (!S.get('cinemaMode') || !panel.classList.contains('dim')) return;
                const r = panel.getBoundingClientRect();
                if (e.clientX >= r.left - 24 && e.clientX <= r.right + 24 && e.clientY >= r.top - 24 && e.clientY <= r.top + 80) {
                    panel.classList.remove('dim');
                    clearTimeout(this._dimT);
                    this._dimT = setTimeout(() => panel.classList.add('dim'), CFG.dimDelay);
                }
            });

            const minBtn = this.$('nfpMin');
            this._on(minBtn, 'click', () => {
                const m = panel.classList.toggle('min');
                minBtn.textContent = m ? '□' : '−';
                S.set('minimized', m);
                if (m) this.fps.stop();
            });
            this._on(this.$('nfpX'), 'click', () => { this.open = false; this._force = false; S.set('panelOpen', false); this._syncVisibility(); });
            this._on(this.launcher, 'click', () => { this.open = true; this._force = !isWatchPage(); S.set('panelOpen', true); this._syncVisibility(); });
        }

        /* ───────── live stats ───────── */
        _bufferAhead(v) {
            try {
                const t = v.currentTime, b = v.buffered;
                for (let i = 0; i < b.length; i++) {
                    if (t >= b.start(i) - 0.1 && t <= b.end(i)) return b.end(i) - t;
                }
            } catch { /* ignore */ }
            return 0;
        }

        _tick() {
            if (this._dead) return;
            const v = DOM.video();
            const watching = !!v && isWatchPage();
            const playing = watching && !v.paused && !v.ended && v.readyState >= 3;

            this.watch.tick(playing);
            this.sleep.tick(watching ? v : null);
            this.browse.tick();
            if (watching) this._autoLoadSeriesProfile();
            const mbps = this.net.sample();

            if (watching && this.lights && this.lights.classList.contains('paused')) {
                this._updateMarquee(v);
            }

            const statsOpen = this._panelShown() && !this.panel.classList.contains('min') && !!this.panel.querySelector('details[data-sec="stats"]')?.open;
            this.fps.sync(v, watching && statsOpen);

            if (!this._panelShown() || this.panel.classList.contains('min')) return;
            const set = (id, txt) => { const el = this.$(id); if (el && el.textContent !== txt) el.textContent = txt; };

            if (watching) {
                set('nfpRes', v.videoWidth ? `${v.videoWidth}×${v.videoHeight}` : '–');
                set('nfpFps', this.fps.fps > 0 ? `${this.fps.fps.toFixed(2).replace(/\.00$/, '')} fps` : '–');
                set('nfpBuf', `${Math.round(this._bufferAhead(v))}s`);

                let dropped = 0, total = 0;
                if (typeof v.getVideoPlaybackQuality === 'function') {
                    try { const q = v.getVideoPlaybackQuality(); dropped = q.droppedVideoFrames || 0; total = q.totalVideoFrames || 0; }
                    catch { /* ignore */ }
                } else {
                    dropped = v.webkitDroppedFrameCount || 0;
                }
                set('nfpDr', total ? `${dropped}/${total}` : String(dropped));

                const dur = NFPlayer.duration() || (isFinite(v.duration) ? v.duration : 0);
                const cur = NFPlayer.currentTime() || v.currentTime || 0;
                const rem = dur > 0 ? Math.max(0, (dur - cur) / (v.playbackRate || 1)) : 0;
                set('nfpLeft', dur > 0 ? fmtClock(rem) : '–');
                set('nfpEnd', dur > 0 ? fmtWallTime(rem) : '–');

                const a = this.net.avg();
                if (this.net.blocked) { set('nfpNet', 'n/a*'); set('nfpAvg', 'n/a*'); }
                else {
                    set('nfpNet', mbps > 0 ? `${mbps.toFixed(1)} Mbps` : 'idle');
                    set('nfpAvg', a > 0 ? `${a.toFixed(1)} Mbps` : '–');
                }
                const note = this.$('nfpNetNote');
                if (note) note.hidden = !this.net.blocked;

                const mt = this.$('nfpMt');
                if (mt) mt.textContent = v.muted ? '🔊 Unmute' : '🔇 Mute';
                this._syncPinUI();
                if (this.panel.querySelector('details[data-sec="tools"]')?.open) this._syncTrackLists();
            } else {
                ['nfpRes', 'nfpFps', 'nfpNet', 'nfpAvg', 'nfpBuf', 'nfpDr', 'nfpLeft', 'nfpEnd'].forEach(id => set(id, '–'));
                set('nfpPinSt', 'Not on watch page');
            }

            set('nfpToday', fmtDur(this.watch.today()));
            set('nfpH1', fmtDur(this.watch.today()));
            set('nfpH7', fmtDur(this.watch.days(7)));
            set('nfpHT', fmtDur(this.watch.total()));
            set('nfpHS', fmtDur(this.watch.session));
            const ss = S.get('skipStats');
            set('nfpHK', `${ss.count} · ${fmtDur(ss.sec)} saved`);
            set('nfpSleepSt', this.sleep.label());

            const ai = S.get('autoIntro'), an = S.get('autoNext'), pip = !!document.pictureInPictureElement;
            this._pill('nfpPillI', ai, `Skip intro: ${ai ? 'on' : 'off'}`);
            this._pill('nfpPillN', an, `Next episode: ${an ? 'on' : 'off'}`);
            this._pill('nfpPillP', pip, `Mini player: ${pip ? 'on' : 'off'}`);
            const pb = this.$('nfpPip');
            if (pb) pb.classList.toggle('active', pip);
        }

        /* ───────── picture filters ───────── */
        _ensureSvgFilter(gamma, temp, sh) {
            const sig = `${gamma}|${temp}|${sh}`;
            let svg = document.getElementById('nfp-svg');
            if (svg && this._svgSig === sig) return;
            this._svgSig = sig;

            const NS = 'http://www.w3.org/2000/svg';
            if (!svg) {
                svg = document.createElementNS(NS, 'svg');
                svg.id = 'nfp-svg';
                svg.setAttribute('width', '0');
                svg.setAttribute('height', '0');
                svg.setAttribute('aria-hidden', 'true');
                svg.style.position = 'absolute';
                document.body.appendChild(svg);
            }
            while (svg.firstChild) svg.removeChild(svg.firstChild);

            const f = document.createElementNS(NS, 'filter');
            f.id = 'nfp-fx';
            f.setAttribute('color-interpolation-filters', 'sRGB');
            f.setAttribute('x', '0');
            f.setAttribute('y', '0');
            f.setAttribute('width', '1');
            f.setAttribute('height', '1');

            if (gamma !== 100) {
                const exp = (100 / gamma).toFixed(3);
                const ct = document.createElementNS(NS, 'feComponentTransfer');
                ['feFuncR', 'feFuncG', 'feFuncB'].forEach(tag => {
                    const fn = document.createElementNS(NS, tag);
                    fn.setAttribute('type', 'gamma');
                    fn.setAttribute('amplitude', '1');
                    fn.setAttribute('exponent', exp);
                    fn.setAttribute('offset', '0');
                    ct.appendChild(fn);
                });
                f.appendChild(ct);
            }

            if (temp !== 0) {
                const t = temp / 50;
                const r = 1 + 0.10 * t;
                const g = t > 0 ? 1 + 0.02 * t : 1;
                const b = t > 0 ? 1 - 0.14 * t : 1 - 0.12 * t;
                const cm = document.createElementNS(NS, 'feColorMatrix');
                cm.setAttribute('type', 'matrix');
                cm.setAttribute('values', `${r.toFixed(3)} 0 0 0 0 0 ${g.toFixed(3)} 0 0 0 0 0 ${b.toFixed(3)} 0 0 0 0 0 1 0`);
                f.appendChild(cm);
            }

            if (sh > 0) {
                const a = (sh / 100) * 0.85;
                const m = document.createElementNS(NS, 'feConvolveMatrix');
                m.setAttribute('order', '3');
                m.setAttribute('divisor', '1');
                m.setAttribute('preserveAlpha', 'true');
                m.setAttribute('kernelMatrix', `0 ${-a.toFixed(3)} 0 ${-a.toFixed(3)} ${(1 + 4 * a).toFixed(3)} ${-a.toFixed(3)} 0 ${-a.toFixed(3)} 0`);
                f.appendChild(m);
            }

            svg.appendChild(f);
        }

        _applyFilters(v) {
            v = v || DOM.video();
            if (!v) return;
            const { brightness: b, contrast: c, saturation: s, gamma: g, temp: t, sharpness: sh } = this.filters;
            const parts = [];
            if (b !== 100) parts.push(`brightness(${b}%)`);
            if (c !== 100) parts.push(`contrast(${c}%)`);
            if (s !== 100) parts.push(`saturate(${s}%)`);
            if (g !== 100 || t !== 0 || sh > 0) {
                this._ensureSvgFilter(g, t, sh);
                parts.push('url(#nfp-fx)');
            }
            const str = parts.join(' ');
            if (this._fApplied.get(v) === str && !!v.style.filter === !!str) return;
            v.style.filter = str;
            this._fApplied.set(v, str);
        }

        _filterRows() {
            return [
                { sl: 'nfpGam', vl: 'nfpGamV', key: 'gamma', fmt: n => (n / 100).toFixed(2) },
                { sl: 'nfpSh', vl: 'nfpShV', key: 'sharpness', fmt: n => n === 0 ? 'Off' : n + '%' },
                { sl: 'nfpTmp', vl: 'nfpTmpV', key: 'temp', fmt: n => n === 0 ? 'Neutral' : (n > 0 ? `Warm +${n}` : `Cool ${n}`) },
                { sl: 'nfpBri', vl: 'nfpBriV', key: 'brightness', fmt: n => n + '%' },
                { sl: 'nfpCon', vl: 'nfpConV', key: 'contrast', fmt: n => n + '%' },
                { sl: 'nfpSat', vl: 'nfpSatV', key: 'saturation', fmt: n => n + '%' }
            ];
        }

            this._filterRows().forEach(({ sl, vl, key, fmt }) => {

                if (ve) ve.textContent = fmt(this.filters[key]);
            });
            const match = Object.keys(this.presets)
                .find(n => Object.keys(this.presets[n]).every(k => this.presets[n][k] === this.filters[k]));
            const pre = this.$('nfpPre');
            if (pre) pre.value = match || 'Custom';
        }

        _autoCropBars() {
            const v = DOM.video();
            if (!v || !v.videoWidth) { this.toast('⚠️ Play a video first', 'zoom'); return; }
            try {
                const W = 64, H = 180;
                const c = document.createElement('canvas');
                c.width = W; c.height = H;
                const ctx = c.getContext('2d');
                ctx.drawImage(v, 0, 0, W, H);
                const px = ctx.getImageData(0, 0, W, H).data;

                const rowLit = y => {
                    let bright = 0;
                    for (let x = 8; x < W - 8; x += 4) {
                        const i = (y * W + x) * 4;
                        if (px[i] > 15 || px[i + 1] > 15 || px[i + 2] > 15) bright++;
                    }
                    return bright >= 2;
                };

                let top = 0, bot = H - 1;
                while (top < H / 2 && !rowLit(top)) top++;
                while (bot > H / 2 && !rowLit(bot)) bot--;
                const active = bot - top + 1;
                if (active <= 10 || top >= H / 2 - 2) {
                    this.toast('ℹ️ Frame unreadable (DRM or dark scene) — try 2.00:1 or 21:9', 'zoom');
                    return;
                }
                const ratio = clamp(Math.round((H / active) * 100), 100, 150);
                this._setZoom(ratio);
                this.toast(ratio === 100 ? '✨ No letterbox bars detected (100%)' : `✨ Auto-cropped black bars (${ratio}%)`, 'zoom');
            } catch {
                this.toast('ℹ️ Auto-crop blocked by DRM — use 2.00:1 or 21:9 preset', 'zoom');
            }
        }

        /* ───────── subtitles (including Cinema Silver-Warm tint) ───────── */

                el.id = 'nfp-subs-style';
                document.head.appendChild(el);

            const rules = [];
            if (hidden) {
                rules.push('.watch-video .player-timedtext { opacity: 0 !important; pointer-events: none !important; }');
            } else {
                if (lift > 0) {
                    rules.push(`.watch-video .player-timedtext { translate: 0 -${lift}vh !important; }`);
                }
                if (size !== 100) {
                    rules.push(`.watch-video .player-timedtext-text-container { scale: ${(size / 100).toFixed(2)} !important; transform-origin: center bottom !important; }`);
                }
                if (silver) {
                    rules.push('.watch-video .player-timedtext-text-container span { color: rgba(244, 237, 222, 0.92) !important; text-shadow: 0 2px 6px rgba(0,0,0,.95), 0 0 2px rgba(0,0,0,.9) !important; }');
                }

                    rules.push('.watch-video .player-timedtext-text-container span { background: rgba(0,0,0,.72) !important; padding: .08em .36em !important; border-radius: 5px !important; box-decoration-break: clone; -webkit-box-decoration-break: clone; }');
                }
            }
            el.textContent = rules.join('\n');
        }

        _syncSubsUI() {
            const { size, lift, bg, hidden } = this.subs;
            const ss = this.$('nfpSubS'), ssv = this.$('nfpSubSV');
            const sl = this.$('nfpSubL'), slv = this.$('nfpSubLV');
            const cbg = this.$('nfpSubBg'), chid = this.$('nfpSubHid');
            if (ss) ss.value = size;
            if (ssv) ssv.textContent = size + '%';
            if (sl) sl.value = lift;
            if (slv) slv.textContent = lift === 0 ? 'Default' : `+${lift}%`;
            if (cbg) cbg.checked = !!bg;
            if (chid) chid.checked = !!hidden;
        }

        _setSubs(patch) {
            Object.assign(this.subs, patch);
            S.set('subs', Object.assign({}, this.subs));
            this._applySubsCss();
            this._syncSubsUI();
        }

        _renderSubLog() {
            const box = this.$('nfpSubLog');
            if (!box) return;
            const list = this.subTracker.lines;
            if (!list.length) {
                setHTML(box, '<span class="nfp-note" style="margin:0">Subtitles will appear here while playing…</span>');
                return;
            }
            const html = list.slice().reverse().map((it, idx) => {
                const safe = it.text.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
                return `<button type="button" class="nfp-subitem" data-subidx="${list.length - 1 - idx}"><span class="nfp-subt">${fmtClock(it.t)}</span><span>${safe}</span></button>`;
            }).join('');
            setHTML(box, html);
        }

        _replaySub() {
            const item = this.subTracker.replay();
            if (!item) { this.toast('ℹ️ No recent subtitle line recorded yet', 'subs'); return; }
            this.toast(`⏪ ${fmtClock(item.t)} · ${item.text.slice(0, 42)}`, 'subs');
        }

        _copySub() {
            const txt = this.subTracker.currentText() || (this.subTracker.lines[this.subTracker.lines.length - 1]?.text ?? '');
            if (!txt) { this.toast('ℹ️ No subtitle text to copy', 'subs'); return; }
            if (navigator.clipboard) {
                navigator.clipboard.writeText(txt).then(() => this.toast('📋 Copied subtitle line', 'subs')).catch(() => this.toast('⚠️ Clipboard blocked', 'subs'));
            }
        }

                document.head.appendChild(el);

            .episodeSelector-item .titleCard-synopsis, [data-uia="episode-item"] p, .watch-video [data-uia="evidence-overlay-synopsis"] { filter: blur(7px) !important; transition: filter .25s ease !important; }
            .episodeSelector-item:hover .titleCard-imageWrapper img, [data-uia="episode-item"]:hover img,

        _syncPinUI() {
            const st = this.$('nfpPinSt');
            if (!st) return;
            const name = DOM.seriesName();
            if (!name) { st.textContent = 'Detecting title…'; return; }

        _saveSeriesProfile() {
            const name = DOM.seriesName();
            if (!name) { this.toast('⚠️ Play a title first so its name can be detected', 'pin'); return; }
            const map = Object.assign({}, S.get('showProfiles') || {});
            map[name] = {
                speed: S.get('speed'),
                zoom: S.get('zoom'),
                filters: Object.assign({}, this.filters),
                audioMode: S.get('audioMode'),
                cinemaMode: S.get('cinemaMode'),
                subLift: this.subs.lift
            };
            S.set('showProfiles', map);
            this._loadedSeries = name;
            this._syncPinUI();
            this.toast(`📌 Saved profile for ${name}`, 'pin');
        }

        _clearSeriesProfile() {
            const name = DOM.seriesName();
            const map = Object.assign({}, S.get('showProfiles') || {});
            if (!name || !map[name]) { this.toast('ℹ️ No saved profile for this show', 'pin'); return; }
            delete map[name];
            S.set('showProfiles', map);
            this._syncPinUI();
            this.toast(`🗑️ Removed profile for ${name}`, 'pin');
        }

        _autoLoadSeriesProfile() {
            const name = DOM.seriesName();
            if (!name || name === this._loadedSeries) return;
            this._loadedSeries = name;
            const p = (S.get('showProfiles') || {})[name];
            if (!p) return;
            if (typeof p.speed === 'number') this._setSpeed(p.speed);
            if (typeof p.zoom === 'number') this._setZoom(p.zoom);
            if (p.filters) this._setFilters(p.filters);
            if (typeof p.subLift === 'number') this._setSubs({ lift: p.subLift });
            if (typeof p.cinemaMode === 'boolean') S.set('cinemaMode', p.cinemaMode);
            if (p.audioMode) this._setAudio(S.get('boost'), p.audioMode);
            this._applyCinema();
            this.toast(`📌 Loaded profile: ${name}`, 'pin');
        }

        /* ───────── speed ───────── */
        _applySpeed(v) {
            v = v || DOM.video();
            if (!v) return;
            const sp = S.get('speed');
            try { v.preservesPitch = true; } catch { /* ignore */ }
            if (Math.abs(v.playbackRate - sp) > 0.001) v.playbackRate = sp;
        }

        _syncSpeedUI() {
            const sp = S.get('speed');
            const sl = this.$('nfpSp'), sv = this.$('nfpSpV');
            if (sl) sl.value = sp;
            if (sv) sv.textContent = sp.toFixed(2) + 'x';
            this.panel.querySelectorAll('[data-speed]')
                .forEach(b => b.classList.toggle('active', Math.abs(parseFloat(b.dataset.speed) - sp) < 0.001));
        }

        _setSpeed(val) {
            val = clamp(Math.round(val * 20) / 20, 0.25, 3);
            S.set('speed', val);
            this._applySpeed();
            this._syncSpeedUI();
            return val;
        }

        /* ───────── audio boost & profiles ───────── */
        _syncAudioUI() {
            const b = S.get('boost'), m = S.get('audioMode');
            const sl = this.$('nfpBoost'), sv = this.$('nfpBoostV'), sel = this.$('nfpAudM');
            if (sl) sl.value = b;
            if (sv) sv.textContent = b + '%';
            if (sel) sel.value = m;
        }

        async _setAudio(pct, mode) {
            pct = clamp(Math.round(pct / 10) * 10, 100, 300);
            mode = AUDIO_PRESETS[mode] ? mode : 'off';
            S.set('boost', pct);
            S.set('audioMode', mode);
            this._syncAudioUI();
            const res = await this.audio.apply(pct, this._effectiveAudioMode(), DOM.video());
            if (res === false) {
                S.set('boost', 100);
                S.set('audioMode', 'off');
                this._syncAudioUI();
                this.toast('⚠️ Web Audio enhancement isn\'t available on this page', 'boost');
            } else if (res === 'pending') {
                this._armGesture('audio', () => this._setAudio(S.get('boost'), S.get('audioMode')));
            }
            return pct;
        }

        _armGesture(flag, fn) {
            if (this._gestureArmed[flag] || this._dead) return;
            this._gestureArmed[flag] = true;
            const h = () => {
                document.removeEventListener('pointerdown', h, true);
                document.removeEventListener('keydown', h, true);
                this._gestureArmed[flag] = false;
                if (!this._dead) fn();
            };
            document.addEventListener('pointerdown', h, true);
            document.addEventListener('keydown', h, true);
        }

        /* ───────── A-B loop ───────── */
        _syncAbUI() {
            const st = this.$('nfpAbSt'), ba = this.$('nfpAbA'), bb = this.$('nfpAbB');
            const { a, b } = this.ab;
            if (ba) { ba.textContent = a !== null ? `A: ${fmtClock(a)}` : 'Set A'; ba.classList.toggle('active', a !== null); }
            if (bb) { bb.textContent = b !== null ? `B: ${fmtClock(b)}` : 'Set B'; bb.classList.toggle('active', b !== null); }
            if (st) st.textContent = (a !== null && b !== null) ? `${fmtClock(a)} → ${fmtClock(b)}` : (a !== null ? `A=${fmtClock(a)}` : 'Off');
        }

        _setAb(which) {
            if (which === 'clear') {
                this.ab = { a: null, b: null };
                this._syncAbUI();
                this.toast('🔁 A–B loop cleared', 'ab');
                return;
            }
            const t = NFPlayer.currentTime() || (DOM.video()?.currentTime ?? 0);
            if (which === 'a') {
                this.ab.a = t;
                if (this.ab.b !== null && this.ab.b <= t + 1) this.ab.b = null;
                this.toast(`🔁 Loop A set at ${fmtClock(t)}`, 'ab');
            } else if (which === 'b') {
                if (this.ab.a === null) this.ab.a = Math.max(0, t - 5);
                if (t <= this.ab.a + 1) { this.toast('⚠️ Point B must be at least 1s after A', 'ab'); return; }
                this.ab.b = t;
                this.toast(`🔁 Looping ${fmtClock(this.ab.a)} → ${fmtClock(t)}`, 'ab');
            }
            this._syncAbUI();
        }

        /* ───────── browse-page autoplay preview muter ───────── */
        _enforceBrowseMute() {
            const mute = S.get('muteBrowse');
            if (isWatchPage()) return;
            document.querySelectorAll('video').forEach(v => {
                if (mute) {
                    if (!v.muted) { v.muted = true; this._autoMuted.add(v); }
                } else if (this._autoMuted.has(v)) {
                    v.muted = false;
                    this._autoMuted.delete(v);
                }
            });
        }

        _wireBrowseMute() {
            const handler = e => {
                if (!S.get('muteBrowse') || isWatchPage()) return;
                const v = e.target;
                if (v && v.tagName === 'VIDEO' && !v.muted) {
                    v.muted = true;
                    this._autoMuted.add(v);
                }
            };
            this._on(document, 'play', handler, true);
            this._on(document, 'volumechange', handler, true);
        }

        /* ───────── actions ───────── */
        _seek(seconds) {
            if (this._seeking) return;
            this._seeking = true;
            this.skip.suspend(2000);
            if (NFPlayer.seekBy(seconds)) {
                this.toast(`${seconds > 0 ? '⏩ +' : '⏪ '}${seconds}s`, 'seek');
            } else {
                warn('Netflix player API not available — seek skipped');
                this.toast('⚠️ Seek unavailable', 'seek');
            }
            setTimeout(() => { this._seeking = false; }, 200);
        }

        _fs() {
            if (DOM.fsRoot()) {
                (document.exitFullscreen || document.webkitExitFullscreen).call(document);
            } else {
                const de = document.documentElement;
                const fn = de.requestFullscreen || de.webkitRequestFullscreen;
                if (fn) fn.call(de);
            }
        }

        _mute() {
            let muted = NFPlayer.call(p => { const m = !p.getMuted(); p.setMuted(m); return m; }, null);
            if (muted === null) {
                const v = DOM.video();
                if (!v) return;
                v.muted = !v.muted;
                muted = v.muted;
            }
            this.toast(muted ? '🔇 Muted' : '🔊 Unmuted', 'mute');
        }

        async _screenshot() {
            const v = DOM.video();
            if (!v || !v.videoWidth) { this.toast('⚠️ No video to capture', 'shot'); return; }
            try {
                const c = document.createElement('canvas');
                c.width = v.videoWidth; c.height = v.videoHeight;
                const ctx = c.getContext('2d');
                ctx.drawImage(v, 0, 0, c.width, c.height);

                const probe = document.createElement('canvas');
                probe.width = 32; probe.height = 18;
                const pctx = probe.getContext('2d');
                pctx.drawImage(c, 0, 0, 32, 18);
                const px = pctx.getImageData(0, 0, 32, 18).data;
                let lit = false;
                for (let i = 0; i < px.length; i += 4) { if (px[i] > 12 || px[i + 1] > 12 || px[i + 2] > 12) { lit = true; break; } }
                if (!lit) { this.toast('⚠️ Frame came out blank — this browser blocks captures of DRM-protected video', 'shot'); return; }

                const blob = await new Promise(res => c.toBlob(res, 'image/png'));
                if (!blob) throw new Error('toBlob returned null');
                const title = DOM.title().replace(/[^\w-]+/g, '_').slice(0, 60);
                const a = document.createElement('a');
                a.href = URL.createObjectURL(blob);
                a.download = `${title}_${fmtClock(NFPlayer.currentTime() || v.currentTime).replace(/:/g, '-')}.png`;
                document.body.appendChild(a);
                a.click();
                a.remove();
                setTimeout(() => URL.revokeObjectURL(a.href), 4000);
                this.toast('📸 Screenshot saved', 'shot');
            } catch (err) {
                warn('Screenshot failed:', err);
                this.toast('⚠️ Screenshot blocked by the browser', 'shot');
            }
        }

        _nfHotkey(letter) {
            const code = letter.charCodeAt(0);
            const init = {
                key: letter,
                code: 'Key' + letter,
                keyCode: code,
                which: code,
                ctrlKey: true,
                altKey: true,
                shiftKey: true,
                bubbles: true,
                cancelable: true,
                composed: true
            };
            const t = document.activeElement || document.body;
            t.dispatchEvent(new KeyboardEvent('keydown', init));
            t.dispatchEvent(new KeyboardEvent('keyup', init));
        }

        _syncTrackLists() {
            const aSel = this.$('nfpAudTrk'), tSel = this.$('nfpTxtTrk');
            if (aSel && document.activeElement !== aSel) {
                const list = NFPlayer.audioTracks();
                const cur = NFPlayer.currentAudio();
                const curId = cur && (cur.trackId ?? cur.id ?? '');
                const sig = list.map(x => x.trackId ?? x.id ?? x.displayName).join('|') + '::' + curId;
                if (aSel.dataset.sig !== sig) {
                    aSel.dataset.sig = sig;
                    setHTML(aSel, list.length
                        ? list.map((t, i) => {
                            const id = t.trackId ?? t.id ?? i;
                            const ch = t.channels ? ` (${t.channels})` : '';
                            const lbl = (t.displayName || t.languageDescription || t.language || `Track ${i + 1}`) + ch;
                            const sel = id === curId ? ' selected' : '';
                            return `<option value="${i}"${sel}>${lbl.replace(/</g, '&lt;')}</option>`;
                        }).join('')
                        : '<option value="">No audio tracks exposed</option>');
                }
            }
            if (tSel && document.activeElement !== tSel) {
                const list = NFPlayer.textTracks();
                const cur = NFPlayer.currentText();
                const curId = cur && (cur.trackId ?? cur.id ?? '');
                const sig = list.map(x => x.trackId ?? x.id ?? x.displayName).join('|') + '::' + curId;
                if (tSel.dataset.sig !== sig) {
                    tSel.dataset.sig = sig;
                    setHTML(tSel, list.length
                        ? list.map((t, i) => {
                            const id = t.trackId ?? t.id ?? i;
                            const cc = t.rawTrackType === 'closedcaptions' || t.isForcedNarrative ? ' [CC/FN]' : '';
                            const lbl = (t.displayName || t.languageDescription || t.language || 'Off') + cc;
                            const sel = id === curId ? ' selected' : '';
                            return `<option value="${i}"${sel}>${lbl.replace(/</g, '&lt;')}</option>`;
                        }).join('')
                        : '<option value="">No subtitle tracks exposed</option>');
                }
            }
        }

        _applyMaxBitrate() {
            if (!S.get('maxBitrate') || !isWatchPage()) return;
            const kbps = NFPlayer.lockMaxBitrate();
            if (kbps) log('Locked max offered video bitrate:', kbps);

        /* ───────── mini player ───────── */
        async _miniPlayer() {
            const v = DOM.video();
            if (!v) { this.toast('⚠️ No video found', 'pip'); return; }
            try {
                if (document.pictureInPictureElement) {
                    this._wantPip = false;
                    await document.exitPictureInPicture();
                    return;
                }
                if (!document.pictureInPictureEnabled) { this.toast('⚠️ Mini player isn\'t supported by this browser', 'pip'); return; }
                v.disablePictureInPicture = false;
                await v.requestPictureInPicture();
                this._wantPip = true;
            } catch (err) {
                warn('Mini player failed:', err);
                this.toast('⚠️ Mini player unavailable right now', 'pip');
            }
        }

        _reattachPip(v) {
            if (!this._wantPip || !v || !document.pictureInPictureEnabled) return;
            if (document.pictureInPictureElement === v) return;
            v.disablePictureInPicture = false;
            v.requestPictureInPicture()
                .then(() => this.toast('🖼️ Mini player followed the new episode', 'pip'))
                .catch(err => {
                    if (err && err.name === 'NotAllowedError') {
                        this.toast('🖼️ Click anywhere to bring the mini player back', 'pip');
                        this._armGesture('pip', () => {
                            const cur = DOM.video();
                            if (this._wantPip && cur && !document.pictureInPictureElement) cur.requestPictureInPicture().catch(() => {});
                        });
                    } else warn('Mini player re-attach failed:', err);
                });
        }

        _bindPip(v) {
            this._onV(v, 'enterpictureinpicture', () => this.toast('🖼️ Mini player opened', 'pip'));
            this._onV(v, 'leavepictureinpicture', () => {
                setTimeout(() => {
                    if (v.isConnected && this._wantPip) { this._wantPip = false; this.toast('🖼️ Mini player closed', 'pip'); }
                }, 50);
            });
        }

            };
            this.panel.querySelectorAll('[data-genre]').forEach(b =>
                this._on(b, 'click', () => openGenre(b.dataset.genre)));
            this._on($('nfpGenGo'), 'click', () => openGenre($('nfpGenIn')?.value));
            this._on($('nfpGenIn'), 'keydown', e => { if (e.key === 'Enter') openGenre(e.target.value); });

            // automation
            [
                ['nfpAI', 'autoIntro', 'Skip intro & recap'],
                ['nfpAN', 'autoNext', 'Auto next episode'],
                ['nfpAWC', 'autoCredits', 'Auto-expand credits'],
                ['nfpAC', 'autoContinue', 'Auto "still watching?"'],
                ['nfpMR', 'modalRatings', 'Title popup ratings'],
                ['nfpAS', 'antiSpoiler', 'Anti-spoiler blur'],
                ['nfpMB', 'muteBrowse', 'Mute browse previews'],
                ['nfpBL', 'browseLauncher', 'Browse page launcher'],
                ['nfpSS', 'sideSeek', 'Side-zone seek']
            ].forEach(([id, key, label]) => {
                this._on($(id), 'change', e => {
                    S.set(key, e.target.checked);
                    if (key === 'muteBrowse') this._enforceBrowseMute();
                    if (key === 'antiSpoiler') this._applySpoilerCss();
                    if (key === 'browseLauncher') this._syncVisibility();
                    this.toast(`${e.target.checked ? '✅' : '⛔'} ${label} ${e.target.checked ? 'on' : 'off'}`, 'auto-' + key);
                    this._tick();
                });
            });

            // sleep timer
            this._on($('nfpSleep'), 'change', e => this._setSleep(e.target.value));

            // stats reset
            this._on($('nfpHR'), 'click', () => {
                if (!window.confirm('Reset all watch-time statistics?')) return;
                this.watch.reset();
                S.set('skipStats', { count: 0, sec: 0 });
                this._tick();
            });

            // collapsible sections remember their state
            this._on(this.panel, 'toggle', e => {
                if (this._syncing || !e.target.dataset || !e.target.dataset.sec) return;
                S.set('sections', Object.assign({}, S.get('sections'), { [e.target.dataset.sec]: e.target.open }));
            }, true);

            this._on($('nfpReset'), 'click', () => this._resetAll());
        }

        _syncAllUI() {
            this._syncing = true;
            const secs = S.get('sections');
            this.panel.querySelectorAll('details[data-sec]').forEach(d => {
                const id = d.dataset.sec;
                d.open = id in secs ? !!secs[id] : !!DEFAULT_OPEN[id];
            });
            this._syncing = false;

            const min = !!S.get('minimized');
            this.panel.classList.toggle('min', min);
            this.$('nfpMin').textContent = min ? '□' : '−';

            this._syncCinemaUI();
            this._syncSpeedUI();
            this._syncZoomUI();
            this._syncAbUI();
            this._syncAudioUI();
            this._syncFilterUI();
            this._syncSubsUI();
            this._syncSleepUI();
            [
                ['nfpAI', 'autoIntro'], ['nfpAN', 'autoNext'], ['nfpAWC', 'autoCredits'],
                ['nfpAC', 'autoContinue'], ['nfpMR', 'modalRatings'], ['nfpAS', 'antiSpoiler'],
                ['nfpMB', 'muteBrowse'], ['nfpBL', 'browseLauncher'], ['nfpSS', 'sideSeek'],
                ['nfpSubAutoCp', 'subCopy'], ['nfpMaxBr', 'maxBitrate']
            ].forEach(([id, key]) => { const el = this.$(id); if (el) el.checked = !!S.get(key); });
        }

        _resetAll() {
            if (!window.confirm('Reset every Netflix Helper Pro+ setting to its default?')) return;
            [
                'cinemaMode', 'cinemaLights', 'cinemaAmbi', 'cinemaSubs', 'cinemaSound',
                'speed', 'zoom', 'filters', 'subs', 'audioMode', 'boost', 'autoIntro', 'autoNext',
                'autoCredits', 'autoContinue', 'muteBrowse', 'antiSpoiler', 'modalRatings',
                'browseLauncher', 'subCopy', 'maxBitrate', 'sideSeek', 'minimized', 'sections', 'pos'
            ].forEach(k => S.set(k, clone(DEFAULTS[k])));
            this.filters = clone(DEFAULTS.filters);
            this.subs = clone(DEFAULTS.subs);
            this.ab = { a: null, b: null };
            this.sleep.set('0');
            this.panel.style.left = ''; this.panel.style.top = ''; this.panel.style.right = '';
            this._applySubsCss();
            this._applySpoilerCss();
            this._syncAllUI();
            const v = DOM.video();
            if (v) { this._applyCinema(v); this._applyFilters(v); this._applyZoom(v); this._applySpeed(v); this.audio.apply(100, 'off', v); }
            this.toast('♻️ Settings reset', 'reset');
        }

        /* ───────── keyboard ───────── */
        _keys() {
            const editable = t => !!t && (t.isContentEditable || ['INPUT', 'TEXTAREA', 'SELECT'].includes(t.tagName));
            const SLEEP = ['0', '15', '30', '45', '60', '90', 'end'];
            const ZOOMS = [100, 113, 134];

            this._on(document, 'keydown', e => {
                if (e.isComposing || editable(e.target)) return;
                if (e.ctrlKey || e.metaKey || e.altKey) return;
                const k = e.key, code = e.code;

                if (k === 'F2') {
                    e.preventDefault();
                    if (!e.repeat) this.toggle();
                    return;
                }
                if (e.target.closest && e.target.closest('.nfp')) return;
                if (!isWatchPage()) return;

                let used = true;
                if (!e.shiftKey && (k === 'g' || k === 'G')) { if (!e.repeat) this.toggleCinema(); }
                else if (!e.shiftKey && (k === 'r' || k === 'R')) { if (!e.repeat) this._replaySub(); }
                else if (e.shiftKey && (k === 'x' || k === 'X')) { if (!e.repeat) this._copySub(); }
                else if (!e.shiftKey && k === '[') this.toast('⏪ ' + this._setSpeed(S.get('speed') - 0.1).toFixed(2) + 'x', 'speed');
                else if (!e.shiftKey && k === ']') this.toast('⏩ ' + this._setSpeed(S.get('speed') + 0.1).toFixed(2) + 'x', 'speed');
                else if (!e.shiftKey && k === '\\') { this._setSpeed(1); this.toast('1.00x', 'speed'); }
                else if (e.shiftKey && code === 'BracketLeft') { if (!e.repeat) this._setAb('a'); }
                else if (e.shiftKey && code === 'BracketRight') { if (!e.repeat) this._setAb('b'); }
                else if (e.shiftKey && code === 'Backslash') { if (!e.repeat) this._setAb('clear'); }
                else if (!e.shiftKey && (k === 'p' || k === 'P')) { if (!e.repeat) this._miniPlayer(); }
                else if (!e.shiftKey && (k === 'z' || k === 'Z')) {
                    if (!e.repeat) {
                        const cur = S.get('zoom');
                        const next = ZOOMS[(ZOOMS.indexOf(cur) + 1) % ZOOMS.length];
                        this._setZoom(next);
                        this.toast(`🔍 Zoom: ${next === 100 ? 'Fit (100%)' : next + '%'}`, 'zoom');
                    }
                }
                else if (e.shiftKey && k === 'ArrowRight') this._seek(CFG.bigSeek);
                else if (e.shiftKey && k === 'ArrowLeft') this._seek(-CFG.bigSeek);
                else if (e.shiftKey && k === 'ArrowUp') this._setAudio(S.get('boost') + 10, S.get('audioMode')).then(p => this.toast(`🔊 Boost ${p}%`, 'boost'));
                else if (e.shiftKey && k === 'ArrowDown') this._setAudio(S.get('boost') - 10, S.get('audioMode')).then(p => this.toast(`🔊 Boost ${p}%`, 'boost'));
                else if (e.shiftKey && (k === 'c' || k === 'C')) {
                    if (!e.repeat) {
                        const hid = !this.subs.hidden;
                        this._setSubs({ hidden: hid });
                        this.toast(hid ? '💬 Subtitles hidden' : '💬 Subtitles visible', 'subs');
                    }
                }
                else if (e.shiftKey && (k === 's' || k === 'S')) { if (!e.repeat) this._screenshot(); }
                else if (e.shiftKey && (k === 't' || k === 'T')) {
                    if (!e.repeat) {
                        const cur = this.sleep.mode ? this.sleep.choice : '0';
                        this._setSleep(SLEEP[(SLEEP.indexOf(cur) + 1) % SLEEP.length]);
                    }
                }
                else used = false;

                if (used) { e.preventDefault(); e.stopImmediatePropagation(); }
            }, true);
        }

        _gestures() {
            this._on(document, 'dblclick', e => {
                if (!S.get('sideSeek') || !isWatchPage()) return;
                const v = DOM.video();
                if (!v) return;
                if (e.target.closest && e.target.closest('.nfp, button, a, input, select, [role="slider"]')) return;
                const r = v.getBoundingClientRect();
                if (e.clientX < r.left || e.clientX > r.right) return;
                if (e.clientY < r.top + r.height * 0.12 || e.clientY > r.bottom - r.height * 0.22) return;
                const x = (e.clientX - r.left) / r.width;
                if (x < 0.3) this._seek(-CFG.seekStep);
                else if (x > 0.7) this._seek(CFG.seekStep);
                else return;
                e.preventDefault();
                e.stopImmediatePropagation();
            }, true);
        }

        /* ───────── video / episode lifecycle ───────── */
        _applyAll(v) {
            v = v || DOM.video();
            if (!v) return;
            if (!isWatchPage()) { this._enforceBrowseMute(); return; }
            this._applyFilters(v);
            this._applyZoom(v);
            this._applySpeed(v);
            this._applyCinema(v);
            this._applyMaxBitrate();
        }

        _onSource(v) {
            this.net.reset();
            NFPlayer.reset();
            this._seeking = false;
            setTimeout(() => { if (!this._dead) this._applyAll(v); }, 600);
        }

        _onEpisodeChange(id) {
            log('Episode change:', id || '(none)');
            this._epId = id;
            this._loadedSeries = '';
            this.ab = { a: null, b: null };
            this._syncAbUI();
            this.subTracker.reset();
            this.net.reset();
            this.skip.newEpisode();
            NFPlayer.reset();
            this._seeking = false;
        }

        _watchVideo() {
            this._iv(() => {
                const v = DOM.video();
                if (v === this._video) return;
                this._clearV();
                this._video = v || null;
                if (!v) { this.fps.stop(); this.ambi.stop(); return; }

                this._onV(v, 'loadstart', () => this._onSource(v));
                this._onV(v, 'loadedmetadata', () => this._applyAll(v));
                this._onV(v, 'playing', () => { this._seeking = false; this._syncHouseLights(v); this._applyAll(v); });
                this._onV(v, 'play', () => this._syncHouseLights(v));
                this._onV(v, 'pause', () => this._syncHouseLights(v));
                this._onV(v, 'seeked', () => this._applySpeed(v));
                this._onV(v, 'error', () => { this._seeking = false; this.skip.suspend(CFG.errorRecovery); });
                this._bindPip(v);

                this._applyAll(v);
                if (isWatchPage()) {
                    this._reattachPip(v);
                    if (this._first) { this._first = false; this.toast(`Netflix Helper Pro+ v${VERSION} ✓`, 'hello'); }
                }
            }, 700);
        }

        _watchPage() {
            let lastHref = '';
            this._iv(() => {
                if (location.href !== lastHref) {
                    lastHref = location.href;
                    const id = videoIdFromUrl();
                    if (id !== this._epId) this._onEpisodeChange(id);
                    this._syncVisibility();
                }
            }, 500);
        }

        /* ───────── boot ───────── */
        init() {
            this._css();
            this._applySubsCss();
            this._applySpoilerCss();
            this._html();

            const pos = S.get('pos');
            if (pos) {
                this.panel.style.left = parseFloat(pos.l) + 'px';
                this.panel.style.top = parseFloat(pos.t) + 'px';
                this.panel.style.right = 'auto';
            }

            this._drag();
            this._controls();
            this._keys();
            this._gestures();
            this._wireBrowseMute();
            this._syncAllUI();
            this._syncVisibility();
            this._enforceBrowseMute();

            this._on(document, 'fullscreenchange', () => { this._mount(); this._clampToViewport(); });
            this._on(document, 'webkitfullscreenchange', () => { this._mount(); this._clampToViewport(); });
            this._on(window, 'pagehide', () => this.watch.flush());
            this._on(document, 'visibilitychange', () => { if (document.hidden) this.watch.flush(); });

            this._watchVideo();
            this._watchPage();
            this._iv(() => this._tick(), CFG.tick);
            this.skip.start();
            this.subTracker.start();

            if (typeof GM_registerMenuCommand === 'function') {
                try {
                    this._menuIds.push(GM_registerMenuCommand('Toggle Netflix Helper Pro+ panel', () => this.toggle()));
                } catch { /* ignore */ }
            }
            log(`v${VERSION} ready`);
        }
    }

    /* ─────────────────────────────────────────────────────────────
       BOOT
    ───────────────────────────────────────────────────────────── */
    function boot() {
        if (!document.head || !document.body) {
            requestAnimationFrame(boot);
            return;
        }
        try { if (win._nfp) win._nfp.destroy(); } catch { /* ignore */ }
        document.querySelectorAll('.nfp, .nfp-launch, .nfp-toasts, .nfp-lights, .nfp-ambi, .nfp-ratebar, #nfp-svg, #nfp-style, #nfp-subs-style, #nfp-spoiler-style').forEach(el => el.remove());

        const ui = new UI();
        win._nfp = ui;
        try { ui.init(); } catch (err) { console.error('[NF+] init failed:', err); }
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot, { once: true });
    else boot();

    window.addEventListener('beforeunload', () => { try { if (win._nfp) win._nfp.destroy(); } catch { /* ignore */ } });
})();
